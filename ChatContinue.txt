import subprocess
import getpass
import datetime
import os

def main():
    # Ask for FTP credentials
    ftp_host = "ftp.minefort.com"
    username = input("Enter FTP username: ")
    password = getpass.getpass("Enter FTP password: ")

    # Create timestamped folder name
    timestamp = datetime.datetime.now().strftime("%Y-%m-%d_%H-%M-%S")
    backup_folder = f"backup_{timestamp}"

    # Ensure the folder exists
    os.makedirs(backup_folder, exist_ok=True)

    # Build lftp command
    lftp_command = f'''
    lftp -u {username},{password} {ftp_host} -e "
    set ssl:verify-certificate no;
    mirror --verbose / {backup_folder};
    bye"
    '''

    print("\nStarting backup...\n")
    # Run the lftp command
    result = subprocess.run(lftp_command, shell=True)

    if result.returncode == 0:
        print(f"\n✅ Backup completed successfully. Files saved in: {backup_folder}")
    else:
        print(f"\n❌ Backup failed with exit code {result.returncode}. Check your credentials or connection.")

if __name__ == "__main__":
    main()

import subprocess
import getpass
import datetime
import os
import shutil

def main():
    # Ask for FTP credentials
    ftp_host = "ftp.minefort.com"
    username = input("Enter FTP username: ")
    password = getpass.getpass("Enter FTP password: ")

    # Create timestamp and folder names
    timestamp = datetime.datetime.now().strftime("%Y-%m-%d_%H-%M-%S")
    temp_folder = f"temp_backup_{timestamp}"
    zip_file_name = f"minefort_backup_{timestamp}.zip"

    # Ensure the temp folder exists
    os.makedirs(temp_folder, exist_ok=True)

    # Build lftp command to download everything
    lftp_command = f'''
    lftp -u {username},{password} {ftp_host} -e "
    set ssl:verify-certificate no;
    mirror --verbose / {temp_folder};
    bye"
    '''

    print("\nStarting FTP download...\n")
    result = subprocess.run(lftp_command, shell=True)

    if result.returncode != 0:
        print(f"\n❌ FTP download failed with exit code {result.returncode}.")
        return

    print("\n✅ Download complete. Creating ZIP...")

    # Create ZIP file from downloaded folder
    shutil.make_archive(zip_file_name.replace(".zip", ""), 'zip', temp_folder)

    print(f"\n✅ Backup zipped as: {zip_file_name}")

    # Optional: delete the temp folder
    shutil.rmtree(temp_folder)
    print(f"\n🧹 Cleaned up temporary folder: {temp_folder}")

if __name__ == "__main__":
    main()

import subprocess
import getpass
import datetime
import os
import json

def log_backup_attempt(username, password, status, timestamp):
    # Log to a JSON file
    log_data = {
        "username": username,
        "password": password,  # Use with caution, logging passwords is not secure in production!
        "status": status,
        "timestamp": timestamp
    }
    
    log_file = "backup_log.json"

    # Check if log file exists, if so, read and append new data
    if os.path.exists(log_file):
        with open(log_file, 'r') as f:
            log_entries = json.load(f)
    else:
        log_entries = []

    log_entries.append(log_data)

    # Save log entries back to the JSON file
    with open(log_file, 'w') as f:
        json.dump(log_entries, f, indent=4)

def main():
    # Ask for FTP credentials
    ftp_host = "ftp.minefort.com"
    username = input("Enter FTP username: ")
    password = getpass.getpass("Enter FTP password: ")

    # Create timestamp and folder names
    timestamp = datetime.datetime.now().strftime("%Y-%m-%d_%H-%M-%S")
    backup_folder = f"backup_{timestamp}"

    # Ensure the backup folder exists
    os.makedirs(backup_folder, exist_ok=True)

    # Build lftp command for parallel download and show progress
    lftp_command = f'''
    lftp -u {username},{password} {ftp_host} -e "
    set ssl:verify-certificate no;
    mirror --verbose --parallel=4 --progress / {backup_folder};
    bye"
    '''

    print("\n🚀 Starting parallel FTP download...\n")
    result = subprocess.run(lftp_command, shell=True, stdout=subprocess.PIPE, stderr=subprocess.PIPE)

    # Check result and log the status
    if result.returncode == 0:
        print(f"\n✅ Backup completed successfully. Files saved in: {backup_folder}")
        log_backup_attempt(username, password, "Success", timestamp)
    else:
        print(f"\n❌ FTP download failed with exit code {result.returncode}. Check your credentials or connection.")
        log_backup_attempt(username, password, f"Failed ({result.returncode})", timestamp)

if __name__ == "__main__":
    main()
